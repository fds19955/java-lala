📌 今日踩坑笔记总结
1. GitLab SSH连接端口问题
问题：SSH连接测试成功，但推送代码时提示Permission denied (publickey,password)
原因：GitLab的SSH端口被设置为2224，而不是默认的22，远程仓库URL没有指定端口

解决
git remote set-url origin ssh://git@49.233.185.174:2224/fds/java-lala.git

2. Git推送分支不存在问题
问题：推送代码时提示src refspec feature/initial-commit does not match any
原因：本地分支feature/initial-commit不存在，或者没有提交任何代码
解决方法：先创建并切换到功能分支，添加并提交代码，再推送

解决
git checkout -b feature/initial-commit
git add .
git commit -m "Initial commit: 提交项目初始代码"
git push -u origin feature/initial-commit

4 公钥私钥设置，免密登录，gitlab操作服务器

第一 首先，本地电脑，

免密  ssh ubuntu@49.233.185.174 -p 22

    1. 准备部署凭证

生成SSH密钥对：打开终端执行
ssh-keygen -t ed25519 -C "1160823732@qq.com"
一路回车默认生成，密钥会保存在~/.ssh/id_ed25519（私钥）和~/.ssh/id_ed25519.pub（公钥）

    2. 配置GitLab项目
将公钥内容复制到GitLab项目的Settings > CI/CD > Variables中，添加名为DEPLOY_SSH_PRIVATE_KEY的变量，值为私钥内容（注意不要包含任何多余空格）


第二 其次，很容易失败

问题：你在GitLab CI/CD中使用的是ED25519私钥，但服务器上只有RSA公钥
你应该在本地电脑上执行添加公钥的命令，而不是在服务器上

本地电脑执行
cat ~/.ssh/id_ed25519.pub | ssh ubuntu@49.233.185.174 "mkdir -p ~/.ssh && cat >> ~/.ssh/authorized_keys && chmod 600 ~/.ssh/authorized_keys && chmod 700 ~/.ssh"

验证服务器是否有ED25519公钥
ssh ubuntu@49.233.185.174
cat ~/.ssh/authorized_keys




# docker-compose.yml - 腾讯云4核8G优化版
version: '3.8'

services:
  # GitLab CE - 适配4核8G配置
  gitlab:
    image: gitlab/gitlab-ce:16.11.8-ce.0
    container_name: article-gitlab
    hostname: '49.233.185.174'  # 必须设置为服务器公网IP
    environment:
      GITLAB_OMNIBUS_CONFIG: |
        # 核心配置
        external_url 'http://49.233.185.174:8929'
        gitlab_rails['gitlab_shell_ssh_port'] = 2224
        
        # 资源优化配置
        puma['worker_processes'] = 2           # 4核环境可增加进程数
        sidekiq['max_concurrency'] = 4         # 增加并发处理能力
        postgresql['max_parallel_workers'] = 2
        postgresql['max_worker_processes'] = 4
        
        # 禁用非必要服务
        prometheus['enable'] = false
        alertmanager['enable'] = false
        node_exporter['enable'] = false
        redis_exporter['enable'] = false
        
        # 性能优化
        gitlab_rails['git_timeout'] = 30
        gitlab_rails['shared_cache_expire'] = 86400
    deploy:
      resources:
        limits:
          cpus: '2.0'      # 最多使用2个CPU核心
          memory: 4096M    # 内存限制4GB
        reservations:
          cpus: '1.0'      # 保证1个CPU核心
          memory: 2048M    # 保证2GB内存
    ports:
      - "8929:8929"
      - "2224:22"
    volumes:
      - /home/ubuntu/gitlab/config:/etc/gitlab
      - /home/ubuntu/gitlab/logs:/var/log/gitlab
      - /home/ubuntu/gitlab/data:/var/opt/gitlab
    restart: unless-stopped

  # GitLab Runner - 启用CI/CD能力
  gitlab-runner:
    image: gitlab/gitlab-runner:latest
    container_name: article-gitlab-runner
    deploy:
      resources:
        limits:
          cpus: '0.5'
          memory: 512M
    volumes:
      - /home/ubuntu/runner/config:/etc/gitlab-runner
      - /var/run/docker.sock:/var/run/docker.sock
    restart: unless-stopped
    depends_on:
      - gitlab

  # MySQL数据库
  mysql:
    image: mysql:8.0
    container_name: article-mysql
    environment:
      MYSQL_ROOT_PASSWORD: fds00913
      MYSQL_DATABASE: article_db
      TZ: Asia/Shanghai
      MYSQL_INNODB_BUFFER_POOL_SIZE: 256M  # 优化内存使用
    deploy:
      resources:
        limits:
          cpus: '1.0'
          memory: 1024M
    ports:
      - "3306:3306"
    volumes:
      - /home/ubuntu/mysql/data:/var/lib/mysql
    restart: unless-stopped

  # Redis缓存
  redis:
    image: redis:alpine
    container_name: article-redis
    command: redis-server --maxmemory 512mb --maxmemory-policy allkeys-lru
    deploy:
      resources:
        limits:
          cpus: '0.5'
          memory: 512M
    ports:
      - "6379:6379"
    restart: unless-stopped
