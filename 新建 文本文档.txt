# docker-compose.yml - 腾讯云4核8G优化版
version: '3.8'

services:
  # GitLab CE - 适配4核8G配置
  gitlab:
    image: gitlab/gitlab-ce:16.11.8-ce.0
    container_name: article-gitlab
    hostname: '49.233.185.174'  # 必须设置为服务器公网IP
    environment:
      GITLAB_OMNIBUS_CONFIG: |
        # 核心配置
        external_url 'http://49.233.185.174:8929'
        gitlab_rails['gitlab_shell_ssh_port'] = 2224
        
        # 资源优化配置
        puma['worker_processes'] = 2           # 4核环境可增加进程数
        sidekiq['max_concurrency'] = 4         # 增加并发处理能力
        postgresql['max_parallel_workers'] = 2
        postgresql['max_worker_processes'] = 4
        
        # 禁用非必要服务
        prometheus['enable'] = false
        alertmanager['enable'] = false
        node_exporter['enable'] = false
        redis_exporter['enable'] = false
        
        # 性能优化
        gitlab_rails['git_timeout'] = 30
        gitlab_rails['shared_cache_expire'] = 86400
    deploy:
      resources:
        limits:
          cpus: '2.0'      # 最多使用2个CPU核心
          memory: 4096M    # 内存限制4GB
        reservations:
          cpus: '1.0'      # 保证1个CPU核心
          memory: 2048M    # 保证2GB内存
    ports:
      - "8929:8929"
      - "2224:22"
    volumes:
      - /home/ubuntu/gitlab/config:/etc/gitlab
      - /home/ubuntu/gitlab/logs:/var/log/gitlab
      - /home/ubuntu/gitlab/data:/var/opt/gitlab
    restart: unless-stopped

  # GitLab Runner - 启用CI/CD能力
  gitlab-runner:
    image: gitlab/gitlab-runner:latest
    container_name: article-gitlab-runner
    deploy:
      resources:
        limits:
          cpus: '0.5'
          memory: 512M
    volumes:
      - /home/ubuntu/runner/config:/etc/gitlab-runner
      - /var/run/docker.sock:/var/run/docker.sock
    restart: unless-stopped
    depends_on:
      - gitlab

  # MySQL数据库
  mysql:
    image: mysql:8.0
    container_name: article-mysql
    environment:
      MYSQL_ROOT_PASSWORD: fds00913
      MYSQL_DATABASE: article_db
      TZ: Asia/Shanghai
      MYSQL_INNODB_BUFFER_POOL_SIZE: 256M  # 优化内存使用
    deploy:
      resources:
        limits:
          cpus: '1.0'
          memory: 1024M
    ports:
      - "3306:3306"
    volumes:
      - /home/ubuntu/mysql/data:/var/lib/mysql
    restart: unless-stopped

  # Redis缓存
  redis:
    image: redis:alpine
    container_name: article-redis
    command: redis-server --maxmemory 512mb --maxmemory-policy allkeys-lru
    deploy:
      resources:
        limits:
          cpus: '0.5'
          memory: 512M
    ports:
      - "6379:6379"
    restart: unless-stopped
