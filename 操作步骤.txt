第一部分 网络先通，启动主节点的k3s

1. 先在腾讯云服务器配置 WireGuard（服务端）
# 安装 WireGuard
sudo apt update
sudo apt install -y wireguard resolvconf

# 生成密钥对
wg genkey | sudo tee /etc/wireguard/private.key
sudo chmod 600 /etc/wireguard/private.key
sudo cat /etc/wireguard/private.key | wg pubkey | sudo tee /etc/wireguard/public.key

# 创建 WireGuard 配置（服务端）
sudo tee /etc/wireguard/wg0.conf > /dev/null <<EOF
[Interface]
PrivateKey = $(sudo cat /etc/wireguard/private.key)
Address = 192.168.2.1/24
ListenPort = 51820
SaveConfig = true
PostUp = iptables -A FORWARD -i wg0 -j ACCEPT; iptables -t nat -A POSTROUTING -o eth0 -j MASQUERADE
PostDown = iptables -D FORWARD -i wg0 -j ACCEPT; iptables -t nat -D POSTROUTING -o eth0 -j MASQUERADE

# 阿里云服务器将作为客户端，先预留配置
# [Peer]
# PublicKey = <阿里云服务器的公钥>
# AllowedIPs = 192.168.2.2/32
EOF

# 启用 IP 转发（重要！）
echo 'net.ipv4.ip_forward=1' | sudo tee -a /etc/sysctl.conf
sudo sysctl -p

# 启动 WireGuard
sudo systemctl enable wg-quick@wg0
sudo systemctl start wg-quick@wg0

# 检查接口
ip addr show wg0

6换服务器，配置阿里云服务器（客户端）

# 安装 WireGuard
sudo apt update
sudo apt install -y wireguard

# 生成密钥对
wg genkey | sudo tee /etc/wireguard/private.key
sudo chmod 600 /etc/wireguard/private.key
sudo cat /etc/wireguard/private.key | wg pubkey | sudo tee /etc/wireguard/public.key


# 创建 WireGuard 配置（客户端）
sudo tee /etc/wireguard/wg0.conf > /dev/null <<EOF
[Interface]
PrivateKey = $(sudo cat /etc/wireguard/private.key)

AllowedIPs = 192.168.2.0/24

EOF


# 启动 WireGuard
sudo systemctl enable wg-quick@wg0
sudo systemctl start wg-quick@wg0

[Interface]
PrivateKey = ${ALI_PRIVATE_KEY}
Address = 192.168.2.2/24

[Peer]
PublicKey = ntjboUfZq+F5keE6NoSGxSfEX9aGIv2WC+DH+bfwbGs=

第二部分 阿里云服务器加入腾讯云集群

1 获取主节点token和配置

sudo cat /var/lib/rancher/k3s/server/node-token

2 阿里云服务器安装K3s agent(多等待会，15分钟)

# 创建rancher目录
sudo mkdir -p /etc/rancher/k3s/

然后安装K3s agent

# 安装K3s agent（使用腾讯云主节点信息）
curl -sfL https://get.k3s.io | \
  INSTALL_K3S_MIRROR=cn \
  INSTALL_K3S_VERSION=v1.27.9+k3s1 \
  K3S_URL=https://192.168.2.1:6443 \
  K3S_TOKEN=<替换为腾讯云服务器的token> \
  sh -s - agent \
    --flannel-iface=wg0 \
    --node-ip=192.168.2.2 \
    --node-name k3s-worker-aliyun \
    --with-node-id

3 安装完成后再配置镜像源

# 等待K3s安装完成
sudo systemctl status k3s-agent

# 现在配置镜像源（安装后目录会自动创建）
sudo tee /etc/rancher/k3s/registries.yaml > /dev/null <<EOF
mirrors:
  docker.io:
    endpoint:
      - "https://registry.cn-hangzhou.aliyuncs.com"
      - "https://docker.mirrors.ustc.edu.cn"
EOF

4 在 GitLab 项目中配置镜像仓库认证，直接拉取私有仓库镜像

    原因：即使脚本中配置了登录账号密码，没有这个，都下载不来镜像。DOCKER_AUTH_CONFIG是Docker 客户端识别的标准环境变量，当这个变量被设置时，Docker 客户端会自动使用其中的认证信息，无需手动运行 docker login。
   步骤：Settings→ CI/CD→ Variables
   key:  DOCKER_AUTH_CONFIG
   value:  如下
{
  "auths": {
    "crpi-qaxlisuwj9fp5gv9.cn-beijing.personal.cr.aliyuncs.com": {
      "username": "nick1191472774",
      "password": "您的密码"
    }
  }
}

 8.1 查看 Kubernetes 资源状态
 # 查看所有 Pod

# 查看所有资源

# 查看服务

8.2验证 MySQL 运行状态

# 查看 MySQL Pod 日志
kubectl logs -n article-system mysql-master-0 --tail=20

# 查看 Pod 事件
kubectl describe pod -n article-system mysql-master-0 | grep -A10 "Events:"

# 检查 Pod 运行时长
kubectl get pod -n article-system mysql-master-0 -o wide

8.3 测试mysql连接

# 获取 root 密码（从 Secret）
MYSQL_PASSWORD=$(kubectl get secret -n article-system mysql-secret -o jsonpath='{.data.root-password}' | base64 -d)
echo "MySQL Root 密码: $MYSQL_PASSWORD"

# 进入 Pod 并连接 MySQL
kubectl exec -it -n article-system mysql-master-0 -- mysql -u root -p$MYSQL_PASSWORD -e "SHOW DATABASES;"

# 或者通过端口转发测试
kubectl port-forward -n article-system svc/mysql-master 33306:3306 &
# 然后可以在本地连接：mysql -h 127.0.0.1 -P 33306 -u root -p

8.4 查看 PVC 和存储
# 查看 PVC
kubectl get pvc -n article-system

# 查看 PV
kubectl get pv

# 查看存储使用
kubectl exec -n article-system mysql-master-0 -- df -h
